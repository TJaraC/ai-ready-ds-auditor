export interface AuditIssue {
  id: string;
  nodeId: string;

  issueType: string;
  offendingValue: string;
  suggestedFix: string;
}

export interface DesignToken {
  id: string;
  name: string;
  type: 'color' | 'typography' | 'spacing' | 'other';
  value: string;
  rawValue: string;
  variableName?: string;
  collectionName?: string;
  groupPath: string[];
}

export interface ComponentSpec {
  id: string;
  name: string;
  key: string;
  description: string;
  variants: string[];
  props: string[];
  usageCount: number;
}

export interface AuditMeta {
  schemaVersion: string;
  chunkCount: number;
  totalBytes: number;
  fileId: string;
  fileName: string;
  scannedAt: string;
  checksum: string;
}

export interface AuditReport {
  schemaVersion: string;
  fileId: string;
  fileName: string;
  scannedAt: string;
  summary: {
    totalIssues: number;
    totalTokens: number;
    totalComponents: number;
    issuesByCategory: Record<string, number>;
    healthScore: number;
  };
  issues: AuditIssue[];

}
